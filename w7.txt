document.addEventListener("DOMContentLoaded", function () {

    // Get section radio buttons
    const sectionRadios = document.querySelectorAll(
        'input[name="section"]'
    );

    // Get required elements
    const sectionDisplay = document.getElementById("sectionDisplay");
    const sectionSelect = document.getElementById("sectionSelect");
    const studentCheckboxes = document.querySelectorAll(
        ".student-checkbox"
    );

    const submitBtn = document.getElementById("submitBtn");
    const presentCount = document.getElementById("presentCount");
    const absentCount = document.getElementById("absentCount");
    const attendancePercent = document.getElementById(
        "attendancePercent"
    );


    // -----------------------------------------
    // Section Radio Button Change
    // Required: while loop
    // -----------------------------------------

    let radioIndex = 0;

    while (radioIndex < sectionRadios.length) {

        sectionRadios[radioIndex].addEventListener(
            "change",
            function () {

                if (this.checked) {

                    sectionDisplay.textContent =
                        "Selected Section: " + this.value;

                    // Update hidden selector as well
                    sectionSelect.value = this.value;
                }

            }
        );

        radioIndex++;
    }


    // -----------------------------------------
    // Calculate Attendance
    // -----------------------------------------

    submitBtn.addEventListener("click", function () {

        let present = 0;

        // Required: FOR LOOP
        for (
            let i = 0;
            i < studentCheckboxes.length;
            i++
        ) {

            if (studentCheckboxes[i].checked) {
                present++;
            }

        }


        // Total number of students
        const totalStudents = studentCheckboxes.length;

        // Required: DO...WHILE LOOP
        let absent = 0;
        let counter = 0;

        do {

            if (counter >= present) {
                absent++;
            }

            counter++;

        } while (counter < totalStudents);


        // Calculate percentage
        let percentage = 0;

        if (totalStudents > 0) {
            percentage = (present / totalStudents) * 100;
        }


        // Display results
        presentCount.textContent = present;
        absentCount.textContent = absent;
        attendancePercent.textContent =
            percentage.toFixed(0) + "%";

    });

});

                            while (

                                if (

                            const correct =
                                questions[i].correct;


                            for (

                                j++

                ) {

                    paymentRadios[

                                if (

                calculateBtn.addEventListener(

                        // FOR LOOP
                        for (
                            let i = 0;
                            i < itemCheckboxes.length;
                            i++
                        ) {

                            if (
                                itemCheckboxes[i]
                                    .checked
                            ) {

                                // Get price
                                const price =
                                    Number(
                                        itemCheckboxes[i]
                                            .dataset
                                            .price
                                    );

                                total += price;


                                // Get item name
                                const label =
                                    itemCheckboxes[i]
                                        .parentElement
                                        .textContent
                                        .trim();

                                selectedItems.push(
