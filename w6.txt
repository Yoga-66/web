document.addEventListener("DOMContentLoaded", function () {

    var form = document.getElementById("userForm");

    var nameField = document.getElementById("name");
    var emailField = document.getElementById("email");
    var ageField = document.getElementById("age");

    var nameError = document.getElementById("nameError");
    var emailError = document.getElementById("emailError");
    var ageError = document.getElementById("ageError");

    var successMessage = document.getElementById("successMessage");
    var successData = document.getElementById("successData");

    form.addEventListener("submit", function (event) {

        event.preventDefault();

        nameError.classList.add("d-none");
        emailError.classList.add("d-none");
        ageError.classList.add("d-none");

        nameField.classList.remove("is-invalid");
        emailField.classList.remove("is-invalid");
        ageField.classList.remove("is-invalid");

        successMessage.classList.add("d-none");

        var name = nameField.value.trim();
        var email = emailField.value.trim();
        var age = ageField.value.trim();

        var isValid = true;

        if (name === "") {
            nameError.classList.remove("d-none");
            nameField.classList.add("is-invalid");
            isValid = false;
        }

        var emailPattern = /^[^\s@]+@[^\s@]+\.com$/i;

        if (!emailPattern.test(email)) {
            emailError.classList.remove("d-none");
            emailField.classList.add("is-invalid");
            isValid = false;
        }

        var ageValue = Number(age);

        if (age === "" || ageValue < 18 || ageValue > 120) {
            ageError.classList.remove("d-none");
            ageField.classList.add("is-invalid");
            isValid = false;
        }

        if (!isValid) {
            return;
        }

        successData.innerHTML =
            "Success! Here is the data:<br>" +
            "Name: " + name + "<br>" +
            "Email: " + email + "<br>" +
            "Age: " + age;

        successMessage.classList.remove("d-none");

        form.reset();

    });

});

document.addEventListener("DOMContentLoaded", function () {

    var canvas = document.getElementById("sketchPad");
    var ctx = canvas.getContext("2d");

    var currentColor = document.getElementById("currentColor");

    var drawing = false;

    ctx.strokeStyle = "black";
    ctx.lineWidth = 3;
    ctx.lineCap = "round";

    canvas.addEventListener("mousedown", function (event) {
        drawing = true;
        ctx.beginPath();
        ctx.moveTo(event.offsetX, event.offsetY);
    });

    canvas.addEventListener("mousemove", function (event) {
        if (drawing) {
            ctx.lineTo(event.offsetX, event.offsetY);
            ctx.stroke();
        }
    });

    canvas.addEventListener("mouseup", function () {
        drawing = false;
    });

    canvas.addEventListener("mouseout", function () {
        drawing = false;
    });

    document.getElementById("redBtn").addEventListener("click", function () {
        ctx.strokeStyle = "red";
        currentColor.textContent = "Red";
    });

    document.getElementById("blueBtn").addEventListener("click", function () {
        ctx.strokeStyle = "blue";
        currentColor.textContent = "Blue";
    });

    document.getElementById("greenBtn").addEventListener("click", function () {
        ctx.strokeStyle = "green";
        currentColor.textContent = "Green";
    });

    document.getElementById("defaultBtn").addEventListener("click", function () {
        ctx.strokeStyle = "black";
        currentColor.textContent = "Black";
    });

    document.getElementById("eraserBtn").addEventListener("click", function () {
        ctx.strokeStyle = "white";
        currentColor.textContent = "Eraser";
    });

    document.getElementById("clearBtn").addEventListener("click", function () {
        ctx.clearRect(0, 0, canvas.width, canvas.height);
        ctx.strokeStyle = "black";
        currentColor.textContent = "Black";
    });

});
